import pandas as pd
import numpy as np
import plotly.express as px
import plotly.graph_objects as bg

# 1. CARREGAMENTO E SANEAMENTO DOS DADOS (DATA CLEANING)
def carregar_e_limpar_dados(filepath):
    # Carregar base primária de dados do questionário
    df = pd.read_csv(filepath)
    
    # Tratamento de valores nulos
    df['renda_familiar'].fillna('Não informado', inplace=True)
    df['horas_estudo'].fillna(df['horas_estudo'].median(), inplace=True)
    
    # Remoção de inconsistências e duplicatas
    df.drop_duplicates(subset=['id_estudante'], keep='first', inplace=True)
    
    # Engenharia de Recursos (Feature Engineering): Índice de Vulnerabilidade Digital
    condicoes = [
        (df['internet'] == 'Wi-Fi') & (df['tem_computador'] == True),
        (df['internet'] == 'Wi-Fi') & (df['tem_computador'] == False),
        (df['internet'] == 'Dados Móveis')
    ]
    escolhas = ['Baixa Vulnerabilidade', 'Média Vulnerabilidade', 'Alta Vulnerabilidade']
    df['nivel_vulnerabilidade_digital'] = np.select(condicoes, escolhas, default='Extrema Vulnerabilidade')
    
    return df

# 2. ANÁLISE EXPLORATÓRIA (EDA) E CRUZAMENTOS
def gerar_relatorio_estatistico(df):
    print("=== RESUMO ESTATÍSTICO DA AMOSTRA JUVE ===")
    print(f"Total de Jovens Analisados: {len(df)}")
    print("\nDistribuição de Vulnerabilidade Digital:")
    print(df['nivel_vulnerabilidade_digital'].value_counts(normalize=True) * 100)
    
    # Agrupamento: Média de Horas de Estudo por Faixa de Renda
    groupby_renda = df.groupby('renda_familiar')['horas_estudo'].agg(['mean', 'median', 'std']).reset_index()
    return groupby_renda

# 3. GERAÇÃO DE GRÁFICOS INTERATIVOS (PLOTLY)
def exportar_visualizacoes_html(df):
    # Gráfico de Barras: Estudo x Renda
    fig_estudo = px.bar(
        df, 
        x='renda_familiar', 
        y='horas_estudo',
        color='nivel_vulnerabilidade_digital',
        title='Horas Dedicadas ao Estudo por Renda Familiar e Acesso Tech',
        labels={'horas_estudo': 'Horas de Estudo / Semana', 'renda_familiar': 'Faixa de Renda'},
        template='plotly_dark'
    )
    
    # Exportação para incorporação direta no GitHub Pages
    fig_estudo.write_html("grafico_estudo_renda.html")
    print("Gráfico exportado com sucesso para HTML interativo!")
